TOPIC 01

Go Basics

Learn variables, constants, conditions, loops, functions and basic Go syntax. 

package main

import "fmt"

func add(a int, b int) int {
    return a + b
}

func main() {

    var name string = "Justin"

        fmt.Println("Adult")
    }

    for i := 1; i <= 5; i++ {
        fmt.Println(i)

    fmt.Println(result)

TOPIC 03

Structs

Structs allow you to create your own data types. Backend applications use structs for users, employees, products and API requests. 

type User struct {

    ID       int    `json:"id"`
    Name     string `json:"name"`
    Email    string `json:"email"`
    Role     string `json:"role"`
    IsActive bool   `json:"is_active"`

}

func main() {

    user := User{
        ID:       1,
        Name:     "Justin",
        Email:    "justin@example.com",
        Role:     "Developer",
        IsActive: true,
    }

    fmt.Println(user.Name)

}

}

    fmt.Println("Sending SMS:", message)

TOPIC 05

Error Handling

Go normally uses returned errors instead of try/catch. 

func divide(a, b int) (int, error) {

    if b == 0 {

        return 0,
            errors.New(
                "cannot divide by zero",
            )
    }

    return a / b, nil
}

result, err := divide(10, 2)

if err != nil {

    fmt.Println("Error:", err)

    return
}

fmt.Println(result)
TOPIC 06

Pointers

A pointer stores the memory address of a value.

&

Gets the address.

*

Dereferences the pointer.

number := 100

pointer := &number

fmt.Println(number)

fmt.Println(pointer)

fmt.Println(*pointer)

*pointer = 200

fmt.Println(number)
TOPIC 07

Methods & Receivers

A method is a function attached to a type. 

type User struct {

    Name string

}

func (u User) GetName() string {

    return u.Name

}

func (u *User) SetName(name string) {

    u.Name = name

}

user := User{
    Name: "Justin",
}

fmt.Println(user.GetName())

user.SetName("Justin Rajadurai")

fmt.Println(user.GetName())
TOPIC 08

Goroutines

Goroutines allow functions to execute concurrently. 

func worker(id int) {

    fmt.Println("Worker started:", id)

    time.Sleep(
        500 * time.Millisecond,
    )

    fmt.Println(
        "Worker finished:",
        id,
    )
}

go worker(1)

go worker(2)

go worker(3)

TOPIC 09

Channels

Channels allow goroutines to communicate safely. 

func worker(ch chan string) {

    ch <- "Notification completed"

}

ch := make(chan string)

go worker(ch)

message := <-ch

fmt.Println(message)
TOPIC 10

sync Package

Important synchronization tools include Mutex, RWMutex, WaitGroup and Once. 

var counter int

var mutex sync.Mutex

func increase(
    wg *sync.WaitGroup,
) {

    defer wg.Done()

    for i := 0; i < 1000; i++ {

        mutex.Lock()

        counter++

        mutex.Unlock()
    }
}

var wg sync.WaitGroup

wg.Add(2)

go increase(&wg)

go increase(&wg)

wg.Wait()

fmt.Println(counter)
TOPIC 11

Context

Context is used for cancellation, timeouts, deadlines and request-scoped operations. 

ctx, cancel := context.WithTimeout(
    context.Background(),
    2 * time.Second,
)

defer cancel()

select {

case <-time.After(
    1 * time.Second,
):

    fmt.Println(
        "Operation completed",
    )

case <-ctx.Done():

    fmt.Println(
        "Context cancelled:",
        ctx.Err(),
    )
}

TOPIC 14

HTTP

Go has a powerful built-in HTTP package. 

func helloHandler(
    w http.ResponseWriter,
    r *http.Request,
) {

    w.Header().Set(
        "Content-Type",
        "application/json",
    )

    response := map[string]string{
        "message": "Hello from Go",
    }

    json.NewEncoder(w).Encode(
        response,
    )
}

http.HandleFunc(
    "/hello",
    helloHandler,
)

http.ListenAndServe(
    ":3006",
    nil,
)
